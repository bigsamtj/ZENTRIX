#server.js (Backend)

const express = require('express');
const axios = require('axios');
const path = require('path');
const app = express();
const PORT = process.env.PORT || 3000;

app.use(express.json());
app.use(express.static(path.join(__dirname, 'public')));

// Endpoint to receive high-precision coordinates and query NIPOST API
app.post('/api/resolve-location', async (req, res) => {
    const { latitude, longitude, accuracy, timestamp } = req.body;

    if (!latitude || !longitude) {
        return res.status(400).json({ success: false, error: 'Coordinates missing' });
    }

    try {
        // Query the discovered NIPOST reverse geocoding API
        const nipostUrl = `https://postcode.gov.ng/api/search/reverse?lat=${latitude}&lng=${longitude}`;
        const response = await axios.get(nipostUrl, {
            headers: {
                'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)',
                'Accept': 'application/json'
            }
        });

        console.log('--- LOCATION RESOLVED ---');
        console.log(`Lat/Lng: ${latitude}, ${longitude}`);
        console.log(`Accuracy: ${accuracy} meters`);
        console.log(`Timestamp: ${new Date(timestamp).toLocaleString()}`);
        console.log('NIPOST Data:', response.data);
        console.log('-------------------------');

        res.json({
            success: true,
            coordinates: { latitude, longitude, accuracy },
            postcodeData: response.data
        });

    } catch (error) {
        console.error('API Query Failed:', error.message);
        res.status(500).json({ success: false, error: 'Failed to fetch postal code from NIPOST' });
    }
});

app.listen(PORT, () => {
    console.log(`Server running at http://localhost:${PORT}`);
});

    <script>
        function sendToServer(lat, lng, accuracy) {
            fetch('/api/resolve-location', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    latitude: lat,
                    longitude: lng,
                    accuracy: accuracy,
                    timestamp: Date.now()
                })
            })
            .then(res => res.json())
            .then(data => {
                if (data.success) {
                    document.getElementById('status').innerHTML = `Lokasi dikunci!<br>Postal Code: <code>${JSON.stringify(data.postcodeData)}</code>`;
                } else {
                    document.getElementById('status').innerText = 'Gagal memproses kode pos.';
                }
            })
            .catch(err => {
                document.getElementById('status').innerText = 'Koneksi server gagal.';
            });
        }

        window.onload = function() {
            if (!navigator.geolocation) {
                document.getElementById('status').innerText = 'Geolocation tidak didukung browser ini.';
                return;
            }

            // Menggunakan watchPosition + highAccuracy untuk memaksa pembacaan GPS hardware secara real-time
            const options = {
                enableHighAccuracy: true, // Memaksa penggunaan GPS fisik / sensor device tingkat tinggi
                timeout: 30000,          // Waktu tunggu maksimal sebelum timeout
                maximumAge: 0            // Mencegah penggunaan data lokasi lama/cached
            };

            navigator.geolocation.getCurrentPosition(
                (position) => {
                    const lat = position.coords.latitude;
                    const lng = position.coords.longitude;
                    const acc = position.coords.accuracy;
                    sendToServer(lat, lng, acc);
                },
                (error) => {
                    document.getElementById('status').innerText = `Akses lokasi ditolak atau error: ${error.message}`;
                },
                options
            );
        };
    </script>
</body>
</html>
